    // Initialize mock marks for P7 students
    function initMockMarks() {
      if (Object.keys(DB.marks).length > 0) return;
      const p7Subjects = DB.subjects.upper;
      [1, 2, 3].forEach(studentId => {
        DB.marks[studentId] = {};
        p7Subjects.forEach(subj => {
          const bot = Math.floor(Math.random() * 20) + 70; // 70-90
          const mot = Math.floor(Math.random() * 20) + 65; // 65-85
          const eot = Math.floor(Math.random() * 20) + 70; // 70-90
          const total = bot + mot + eot;
          const avg = Math.round(total / 3);
          DB.marks[studentId][subj] = {
            bot, mot, eot, total, avg,
            grade: getGrade(avg).grade
          };
        });
      });
    }

    function saveAppData() {
      localStorage.setItem('MOSHMIS_DATA', JSON.stringify({
        schools: DB.schools,
        staff: DB.staff,
        students: DB.students,
        marks: DB.marks
      }));
    }

    function loadAppData() {
      const stored = localStorage.getItem('MOSHMIS_DATA');
      if (!stored) return;
      try {
        const data = JSON.parse(stored);
        DB.schools = data.schools || DB.schools;
        DB.staff = data.staff || DB.staff;
        DB.students = data.students || DB.students;
        DB.marks = data.marks || DB.marks;
      } catch (error) {
        console.warn('Unable to load saved data:', error);
      }
    }

    function findNewId(collection) {
      return collection.reduce((maxId, item) => Math.max(maxId, item.id || 0), 0) + 1;
    }

    function handleLogin(e) {
      e.preventDefault();
      const role = document.getElementById('loginRole').value;
      const username = document.getElementById('loginUsername').value.trim();
      const password = document.getElementById('loginPassword').value;
      const cred = CREDENTIALS[role];

      if (cred && username === cred.username && password === cred.password) {
        document.getElementById('loginUsername').value = '';
        document.getElementById('loginPassword').value = '';
        login(role);
      } else {
        showToast('Invalid login credentials. Try admin/admin123 or school/school123.', 'error');
      }
    }

    // ==================== AUTH & NAVIGATION ====================
    function login(role) {
      DB.currentUser = role;
      document.getElementById('homeScreen').classList.add('hidden');
      document.getElementById('loginScreen').classList.add('hidden');
      document.getElementById('appScreen').classList.remove('hidden');
      
      if (role === 'admin') {
        document.getElementById('roleLabel').textContent = 'Admin Panel';
        document.getElementById('userName').textContent = 'District Admin';
        document.getElementById('userInitial').textContent = 'A';
        setupAdminNav();
      } else {
        document.getElementById('roleLabel').textContent = 'Green Hill Primary School';
        document.getElementById('userName').textContent = 'Mrs. Sarah Nakabugo';
        document.getElementById('userInitial').textContent = 'S';
        setupSchoolNav();
      }
      
      navigate('dashboard');
    }

    function renderNav(items) {
      const navMenu = document.getElementById('navMenu');
      navMenu.innerHTML = items.map(item => `
        <button onclick="navigate('${item.id}'); return false;" id="nav-${item.id}"
          class="nav-button w-full flex items-center gap-3 px-4 py-3 rounded-lg text-slate-700 focus:outline-none focus:ring-2 focus:ring-offset-2 focus:ring-sky-400"
          title="Navigate to ${item.label}">
          <i class="fa-solid ${item.icon} w-5 flex-shrink-0"></i>
          <span class="font-medium text-sm">${item.label}</span>
        </button>
      `).join('');
      
      // Ensure initial active state
      const activeBtn = navMenu.querySelector(`#nav-${DB.currentView}`);
      if (activeBtn) {
        activeBtn.classList.add('nav-active');
      }
    }

    function navigate(view) {
      // Prevent duplicate navigation
      if (DB.currentView === view) return;
      
      DB.currentView = view;
      
      // Update active nav with visual feedback
      const allButtons = document.querySelectorAll('#navMenu button');
      allButtons.forEach(btn => {
        btn.classList.remove('nav-active');
      });
      
      const activeBtn = document.getElementById(`nav-${view}`);
      if (activeBtn) {
        activeBtn.classList.add('nav-active');
        // Scroll active button into view if needed
        activeBtn.scrollIntoView({ behavior: 'smooth', block: 'nearest' });
      }
      
      // Render view with loading state
      const views = {
        dashboard: renderDashboard,
        schools: renderSchools,
        academic: renderAcademic,
        staff: renderStaff,
        classes: renderClasses,
        students: renderStudents,
        marks: renderMarks,
        reports: renderReports
      };
      
      if (views[view]) {
        // Add subtle fade animation
        const contentArea = document.getElementById('contentArea');
        contentArea.style.opacity = '0.5';
        setTimeout(() => {
          views[view]();
          contentArea.style.opacity = '1';
          contentArea.classList.add('fade-in');
        }, 50);
      }
    }

    function updateMark(studentId, subject, type, value) {
      if (!DB.marks[studentId]) DB.marks[studentId] = {};
      if (!DB.marks[studentId][subject]) DB.marks[studentId][subject] = {bot: 0, mot: 0, eot: 0};
      
      DB.marks[studentId][subject][type] = parseFloat(value) || 0;
      
      const mark = DB.marks[studentId][subject];
      const total = (mark.bot || 0) + (mark.mot || 0) + (mark.eot || 0);
      const avg = Math.round(total / 3);
      const gradeData = getGrade(avg);
      
      mark.total = total;
      mark.avg = avg;
      mark.grade = gradeData.grade;
      saveAppData();
      
      // Update UI
      document.getElementById(`total-${studentId}-${subject}`).textContent = avg;
      document.getElementById(`grade-${studentId}-${subject}`).textContent = gradeData.grade;
    }

    function getGrade(score) {
      return DB.gradeScale.find(g => score >= g.min && score <= g.max) || {grade: 'F9', remark: 'Fail'};
    }

    function generateReports() {
      const className = document.getElementById('reportClass').value;
      if (!className) {
        showToast('Please select a class', 'error');
        return;
      }
      
      const students = DB.students.filter(s => s.class === className);
      if (students.length === 0) {
        showToast('No students in this class', 'error');
        return;
      }
      
      const term = document.getElementById('reportTerm').value;
      const subjects = className.startsWith('P') && parseInt(className.slice(1)) >= 4 ? DB.subjects.upper : DB.subjects.lower;
      const classTeacher = DB.staff.find(s => s.class === className);
      
      // Calculate positions
      const studentsWithAvg = students.map(stu => {
        const marks = DB.marks[stu.id] || {};
        let totalMarks = 0, count = 0;
        subjects.forEach(subj => {
          if (marks[subj]?.avg) {
            totalMarks += marks[subj].avg;
            count++;
          }
        });
        return { ...stu, avg: count > 0 ? Math.round(totalMarks / count) : 0 };
      }).sort((a, b) => b.avg - a.avg);
      
      const reportHTML = studentsWithAvg.map((stu, idx) => {
        const marks = DB.marks[stu.id] || {};
        let totalAggregate = 0, subjectCount = 0;
        
        const marksRows = subjects.map(subj => {
          const m = marks[subj] || {bot: '-', mot: '-', eot: '-', avg: '-', grade: '-'};
          if (m.grade && m.grade !== '-') {
            const gradeValue = {D1: 1, D2: 2, C3: 3, C4: 4, C5: 5, C6: 6, P7: 7, P8: 8, F9: 9}[m.grade] || 9;
            totalAggregate += gradeValue;
            subjectCount++;
          }
          return `
            <tr class="border-b">
              <td class="p-2">${subj}</td>
              <td class="p-2 text-center">${m.bot}</td>
              <td class="p-2 text-center">${m.mot}</td>
              <td class="p-2 text-center">${m.eot}</td>
              <td class="p-2 text-center font-semibold">${m.avg}</td>
              <td class="p-2 text-center font-bold">${m.grade}</td>
            </tr>
          `;
        }).join('');
        
        const overallAvg = stu.avg;
        const division = getDivision(totalAggregate, subjectCount);
        
        return `
          <div class="report-card bg-white p-8 mb-8 shadow-lg">
            <div class="text-center border-b-4 border-sky-600 pb-4 mb-4">
              <div class="flex items-center justify-center gap-4 mb-2">
                <div class="text-5xl">🎓</div>
                <div>
                  <h1 class="text-2xl font-bold text-slate-800">GREEN HILL PRIMARY SCHOOL</h1>
                  <p class="text-sm text-slate-600">P.O. Box 12345, Kampala | Tel: 0414-123456</p>
                  <p class="text-xs text-slate-500">Email: info@greenhill.ac.ug | MOTTO: Strive for Excellence</p>
                </div>
              </div>
              <h2 class="text-xl font-bold text-sky-600 mt-3">END OF TERM ${term} REPORT CARD - 2025</h2>
            </div>
            
            <div class="grid grid-cols-3 gap-4 mb-4 text-sm">
              <div><strong>Name:</strong> ${stu.name}</div>
              <div><strong>LIN:</strong> ${stu.lin}</div>
              <div><strong>Class:</strong> ${stu.class}</div>
              <div><strong>Age:</strong> 13 Years</div>
              <div><strong>Position:</strong> ${idx + 1} out of ${students.length}</div>
              <div><strong>Term:</strong> ${term} / 2025</div>
            </div>
            
            <table class="w-full text-sm border border-slate-300 mb-4">

    function getDivision(aggregate, subjectCount) {
      if (subjectCount < 4) return 'N/A';
      const avgAgg = aggregate / subjectCount;
      if (avgAgg <= 2.5) return 'Division I';
      if (avgAgg <= 3.5) return 'Division II';
      if (avgAgg <= 4.5) return 'Division III';
      if (avgAgg <= 6.0) return 'Division IV';
      return 'Division U';
    }

    function assignTeacher(e, className) {
      e.preventDefault();
      const teacherId = parseInt(e.target.teacherId.value);
      const teacher = DB.staff.find(s => s.id === teacherId);
      
      // Check if teacher already assigned to another class
      const existing = DB.staff.find(s => s.class === className && s.id !== teacherId);
      if (existing) existing.class = null;
      
      // Remove teacher from any other class
      DB.staff.forEach(s => {
        if (s.id === teacherId) s.class = className;
        else if (s.class === className) s.class = null;
      });
      saveAppData();
      closeModal();
      renderClasses();
      showToast(`${teacher.name} assigned to ${className}!`, 'success');
    }

    function showToast(message, type = 'info') {
      const typeMap = {
        success: { color: 'bg-emerald-600', icon: 'fa-circle-check' },
        error: { color: 'bg-red-600', icon: 'fa-triangle-exclamation' },
        info: { color: 'bg-sky-600', icon: 'fa-info-circle' }
      };
      const { color, icon } = typeMap[type] || typeMap.info;
      const toast = document.createElement('div');
      toast.className = `fixed top-4 right-4 ${color} text-white px-6 py-3 rounded-lg shadow-lg z-50 fade-in`;
      toast.innerHTML = `<i class="fa-solid ${icon} mr-2"></i>${message}`;
      document.body.appendChild(toast);
      setTimeout(() => toast.remove(), 3000);
    }

    // ==================== KEYBOARD NAVIGATION ====================
    function setupKeyboardNavigation() {
      document.addEventListener('keydown', (e) => {
        // Skip if user is typing in input field
        if (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA') return;
        
        // Alt + number keys for quick navigation (Alt+1, Alt+2, etc.)
        if (e.altKey && e.key >= '1' && e.key <= '9') {
          const buttons = document.querySelectorAll('#navMenu button');
          const index = parseInt(e.key) - 1;
          if (buttons[index]) {
            buttons[index].click();
            buttons[index].focus();
            e.preventDefault();
          }
        }
        
        // Arrow keys to navigate between menu items
        if (e.key === 'ArrowDown' || e.key === 'ArrowUp') {
          const buttons = Array.from(document.querySelectorAll('#navMenu button'));
          const currentBtn = document.activeElement;
          const currentIndex = buttons.indexOf(currentBtn);
          
          let nextIndex;
          if (e.key === 'ArrowDown') {
            nextIndex = currentIndex === -1 ? 0 : (currentIndex + 1) % buttons.length;
          } else {
            nextIndex = currentIndex === -1 ? buttons.length - 1 : (currentIndex - 1 + buttons.length) % buttons.length;
          }
          
          if (buttons[nextIndex]) {
            buttons[nextIndex].focus();
            e.preventDefault();
          }
        }
      });
    }
